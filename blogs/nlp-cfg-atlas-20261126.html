02 / 语法字段与程序映射
JSON 字段实际 Python 行为


model.d_model / heads / kv_headsEmbedding 宽度、Q/K/V 投影及 head reshape；验证整除条件。


model.seq_len输入长度上限；不是训练步数或训练预算。


model.layers[].mixer.typev12 MIXERS 分派到 Attention / ShortConv / GatedDelta / Mamba2 / Hyena / RWKV7；none 不构造 mixer。


model.layers[].mlpv12 SwiGLU：hidden 或合法 mult；v16 ReLU²：d_model × mult。


model.ngramv12 可选 Bigram 表与线性投影；仅支持已实现的二元记忆协议。


model.layers[].norm.positionv16 每层 pre / post RMSNorm；v12 不支持此字段。


optim优化器元数据；模型导出不启动训练器。下面 CPU pilot 使用 AdamW。


v12 和 v16 当前不等于 v1.3.1 的完整搜索空间：不支持的字段会报错，而非悄悄忽略或假装复刻。

"""Deterministic algorithmic teachers for a shared autoregressive token interface.

Public synthetic-only data. Prefix tokens are conditioning, never scored unless
explicitly in the target mask. Each Env freezes its teacher, split seeds and
sequence shape. No production queue integration or historical score reuse.
"""
from dataclasses import dataclass, asdict
import hashlib
import json
import numpy as np

VOCAB_SIZE=1024
BOS=900
QUERY=901
ANSWER=902
END=903
IMAGE=904
AUDIO=905
FAMILIES=("markov","delayed_copy","associative_recall","stack","arithmetic","automaton","image_tokens","audio_tokens")
VERSION="nlp-synthetic-teachers-v1"

@dataclass(frozen=True)
class Env:
    family: str
    level: int
    teacher_seed: int=20261126

    def __post_init__(self):
        if self.family not in FAMILIES or type(self.level) is not int or self.level not in (1,2,3):
            raise ValueError("unknown family or level")
        if type(self.teacher_seed) is not int or self.teacher_seed<0:
            raise ValueError("teacher_seed must be nonnegative integer")

    @property
    def seq_len(self):
        return (96,128,192)[self.level-1]

    @property
    def id(self):
        return f"{VERSION}/{self.family}/l{self.level}/teacher{self.teacher_seed}"

def catalog():
    return [Env(f,k) for f in FAMILIES for k in (1,2,3)]

def _teacher_rng(e):
    return np.random.default_rng(np.random.SeedSequence([e.teacher_seed,FAMILIES.index(e.family),e.level,0]))

def _example(e,rng,teacher):
    k=e.level; T=e.seq_len
    prompt=[]; targets=[]; mask=None
    if e.family=="markov":
        v=16*k; seq=[int(rng.integers(1,v+1))]
        for _ in range(T-1):
            seq.append(int(rng.integers(1,v+1)) if rng.random()<.08*k else int(teacher[seq[-1]-1]))
        prompt=[BOS,seq[0]];targets=seq[1:]
    elif e.family=="delayed_copy":
        n=(8,16,32)[k-1]; src=rng.integers(1,64,n).tolist(); distractor=rng.integers(200,264,2*n).tolist()
        prompt=[BOS,*src,*distractor,QUERY,ANSWER];targets=src if k<3 else src[::-1]
    elif e.family=="associative_recall":
        n=(8,16,32)[k-1];keys=rng.choice(np.arange(64,128),n,replace=False).tolist()
        vals=rng.integers(128,192,n).tolist(); q=int(rng.integers(n))
        prompt=[BOS,*[a for kv in zip(keys,vals) for a in kv],QUERY,keys[q],ANSWER];targets=[vals[q]]
    elif e.family=="stack":
        # Typed push/pop; query top after a legal sequence of operations.
        st=[]; stream=[]
        for _ in range((20,40,64)[k-1]):
            if not st or (len(st)<4*k and rng.random()<.65):
                t=int(rng.integers(1,9));st.append(t);stream.extend([910,t])
            else:
                st.pop();stream.append(911)
        prompt=[BOS,*stream,QUERY,ANSWER];targets=[st[-1] if st else 912]
    elif e.family=="arithmetic":
        base=(8,16,32)[k-1]; n=(2,3,4)[k-1]
        a,b=map(int,rng.integers(0,base**n,2))
        digits=lambda x:[1+(x//base**i)%base for i in range(n)]
        # Add modulo base**n, least significant digit first.
        prompt=[BOS,*digits(a),913,*digits(b),ANSWER];targets=digits(a+b)
    elif e.family=="automaton":
        n=4*k; state=int(rng.integers(n)); seq=[BOS,300+state];mask=[False,False]
        for _ in range((24,40,64)[k-1]):
            symbol=int(rng.integers(4));state=int(teacher[state,symbol])
            seq.extend([320+symbol,300+state]);mask.extend([False,True])
        prompt=seq;targets=[]
    elif e.family=="image_tokens":
        # Real synthetic raster, discretized intensities, not natural-image VLM.
        side=(8,9,12)[k-1]; image=np.zeros((side,side),dtype=np.int64)
        shape=int(rng.integers(4)); y=int(rng.integers(1,side-1));x=int(rng.integers(1,side-1))
        if shape==0:image[y,:]=7
        elif shape==1:image[:,x]=7
        elif shape==2:image[y-1:y+2,x-1:x+2]=7
        else:
            for i in range(side):image[i,(i+x)%side]=7
        noise=rng.random(image.shape)<(.035*k)
        image[noise]=rng.integers(0,8,int(noise.sum()))
        prompt=[BOS,IMAGE,*(400+image.ravel()).tolist(),QUERY,ANSWER];targets=[500+shape]
    elif e.family=="audio_tokens":
        n=(64,96,128)[k-1]; label=int(rng.integers(4)); f=(2,4,8,12)[label]
        time=np.arange(n)/n; phase=rng.uniform(0,2*np.pi)
        wave=np.sin(2*np.pi*f*time+phase)+rng.normal(0,.04*k,n)
        bins=np.clip(np.rint((wave+1.2)/2.4*31),0,31).astype(int)
        prompt=[BOS,AUDIO,*(600+bins).tolist(),QUERY,ANSWER];targets=[510+label]
    if mask is None:
        mask=[False]*len(prompt)+[True]*len(targets)
    seq=prompt+targets
    if len(seq)>T+1 or len(seq)!=len(mask) or not any(mask):
        raise AssertionError((e.id,len(seq),T+1))
    if not all(0<=x<VOCAB_SIZE for x in seq):
        raise AssertionError("token out of vocabulary")
    seq=seq+[0]*(T+1-len(seq))
    mask=mask+[False]*(T+1-len(mask))
    return np.asarray(seq,dtype=np.int16), np.asarray(mask[1:],dtype=bool)

def make_splits(env,train_n=512,dev_n=128):
    if min(train_n,dev_n)<1:
        raise ValueError("nonempty splits required")
    tr=_teacher_rng(env)
    teacher=(tr.permutation(16*env.level)+1 if env.family=="markov" else
             tr.integers(0,4*env.level,(4*env.level,4)) if env.family=="automaton" else None)
    out={};seen=set()
    for name,n,split_seed in (("TRAIN",train_n,1),("DEV",dev_n,2)):
        rng=np.random.default_rng(np.random.SeedSequence([env.teacher_seed,FAMILIES.index(env.family),env.level,split_seed]))
        xs=[];ms=[]
        for i in range(n):
            # Exact duplicate sequence rejection across both roles; no shared data.
            for _ in range(1024):
                x,m=_example(env,rng,teacher)
                digest=hashlib.sha256(x.tobytes()+m.tobytes()).digest()
                if digest not in seen:break
            else:raise RuntimeError("teacher support too small for disjoint splits")
            seen.add(digest);xs.append(x);ms.append(m)
        x=np.stack(xs);m=np.stack(ms)
        out[name]={"tokens":x,"target_mask":m}
    return out

def manifest(env,roles):
    return {"version":VERSION,"id":env.id,"teacher":asdict(env),"seq_len":env.seq_len,"vocab_size":VOCAB_SIZE,
            "roles":{r:{"rows":len(d["tokens"]),"supervised_tokens":int(d["target_mask"].sum()),"sha256":hashlib.sha256(d["tokens"].tobytes()+d["target_mask"].tobytes()).hexdigest()} for r,d in roles.items()}}

05 / 10–60 秒搜索预算与准备开销
CPU 生成 / 缓存数据→CPU 校验 / 构建配置→GPU 持久 worker + warm-up→固定训练预算→独立 DEV
可在 CPU 预先完成生成数据、tokenize、打包、校验 shape、生成 Python、计算 key、排队与文件准备。
仍需要 GPU设备传输、CUDA context、实际 kernel warm-up / graph capture；编译缓存仅在版本与设备兼容时复用。
公平性热身后恢复初始权重、优化器、RNG；分别记录 setup、训练、DEV、总时长，不将准备时间隐去。固定数据曝光量与实际耗时分开。
建议先同一硬件测 10 / 30 / 60 秒训练预算，测 cold / warm setup 和排序稳定性；之后选择固定更新数或固定 token 数作为跨硬件协议。未实际完成 GPU 标定，不承诺可达到某个吞吐。
